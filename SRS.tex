\chapter{Đặc tả Tính năng AI (AI Specifications)}
\section{CV-Job Matching Engine}
Tính năng AI Matching sử dụng mô hình ngôn ngữ để kết nối CV và JD dựa trên ngữ nghĩa chuyên môn.

\subsection{Đầu vào (Input Data)}
Hệ thống xử lý văn bản JD (Yêu cầu kỹ thuật, kinh nghiệm) và văn bản CV (Kỹ năng, dự án).

\subsection{Mô hình và Thuật toán}
\begin{itemize}
    \item \textbf{Model}: Fine-tuned \texttt{gemma-it-matching-best}.
    \item \textbf{Logic}: Chuyển đổi văn bản thành Vector (Embedding) và tính toán \textbf{Cosine Similarity}:
    \begin{equation}
        \text{Score} = \frac{V_{JD} \cdot V_{CV}}{\|V_{JD}\| \|V_{CV}\|}
    \end{equation}
\end{itemize}

\chapter{Yêu cầu Phi chức năng (Non-Functional Requirements)}
\section{Hiệu năng (Performance)}
\begin{itemize}
    \item Tìm kiếm từ khóa cơ bản: $< 1.0$ giây.
    \item Tìm kiếm Vector (AI Search): $< 2.5$ giây.
    \item Tải gợi ý trang chủ: $< 3.0$ giây.
\end{itemize}

\section{Bảo mật (Security)}
\begin{itemize}
    \item Xác thực: JWT (JSON Web Token) với cơ chế Refresh Token Rotation.
    \item Phân quyền: Role-based Access Control (RBAC).
    \item Mã hóa: Mật khẩu được băm bằng thuật toán \textbf{BCrypt}.
\end{itemize}

\section{Tính toàn vẹn dữ liệu}
Mọi giao dịch nghiệp vụ nhạy cảm đều được bao bọc bởi annotation \texttt{@Transactional} của Spring để đảm bảo tính Acid.
