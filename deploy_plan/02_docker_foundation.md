# Task 02: Docker Foundation Setup

## Objective

Configure Docker on the EC2 instance, create the application directory structure, set up Docker networks and volumes, and prepare the shared `.env` file with all environment variables.

## Prerequisites

- Task 01 completed (EC2 instance running Ubuntu 24.04 LTS with Docker installed via user-data)
- SSH access to EC2 instance with Elastic IP (username: `ubuntu`)

### 2.6 Create Deployment Scripts

```bash
# Main deployment script
cat > /opt/iting/scripts/deploy.sh << 'DEPLOYEOF'
#!/bin/bash
set -e

COMPOSE_FILE="/opt/iting/docker-compose.yml"
ENV_FILE="/opt/iting/.env"

echo "=== ITing Deployment Script ==="

case "$1" in
  up)
    echo "Starting all services..."
    docker compose -f $COMPOSE_FILE --env-file $ENV_FILE up -d
    echo "Services started. Checking health..."
    sleep 15
    docker compose -f $COMPOSE_FILE --env-file $ENV_FILE ps
    ;;
  down)
    echo "Stopping all services..."
    docker compose -f $COMPOSE_FILE --env-file $ENV_FILE down
    echo "Services stopped."
    ;;
  restart)
    echo "Restarting all services..."
    docker compose -f $COMPOSE_FILE --env-file $ENV_FILE restart
    ;;
  rebuild)
    echo "Rebuilding and restarting all services..."
    docker compose -f $COMPOSE_FILE --env-file $ENV_FILE up -d --build --force-recreate
    ;;
  rebuild-backend)
    echo "Rebuilding backend only..."
    docker compose -f $COMPOSE_FILE --env-file $ENV_FILE up -d --build --force-recreate backend
    ;;
  rebuild-frontend)
    echo "Rebuilding frontend only..."
    docker compose -f $COMPOSE_FILE --env-file $ENV_FILE up -d --build --force-recreate frontend
    ;;
  logs)
    if [ -z "$2" ]; then
      docker compose -f $COMPOSE_FILE --env-file $ENV_FILE logs --tail=100 -f
    else
      docker compose -f $COMPOSE_FILE --env-file $ENV_FILE logs --tail=100 -f "$2"
    fi
    ;;
  status)
    docker compose -f $COMPOSE_FILE --env-file $ENV_FILE ps
    ;;
  health)
    echo "Checking service health..."
    for service in backend nginx-proxy redis kafka prometheus grafana loki tempo; do
      STATUS=$(docker compose -f $COMPOSE_FILE --env-file $ENV_FILE ps $service 2>/dev/null | grep -c "Up" || echo "0")
      if [ "$STATUS" = "1" ]; then
        echo "  ✓ $service: Running"
      else
        echo "  ✗ $service: NOT running"
      fi
    done
    ;;
  *)
    echo "Usage: $0 {up|down|restart|rebuild|rebuild-backend|rebuild-frontend|logs [service]|status|health}"
    exit 1
    ;;
esac
DEPLOYEOF

chmod +x /opt/iting/scripts/deploy.sh

# Backup script
cat > /opt/iting/scripts/backup.sh << 'BACKUPEOF'
#!/bin/bash
set -e

BACKUP_DIR="/opt/iting/backups"
TIMESTAMP=$(date +%Y%m%d_%H%M%S)
RETENTION_DAYS=7

mkdir -p $BACKUP_DIR

# Backup configurations
echo "Backing up configurations..."
tar czf "$BACKUP_DIR/config_$TIMESTAMP.tar.gz" \
  -C /opt/iting config/ monitoring/ .env

# Backup Docker volumes
echo "Backing up volumes..."
tar czf "$BACKUP_DIR/volumes_$TIMESTAMP.tar.gz" \
  -C /var/lib/docker/volumes \
  iting_redis_data iting_prometheus_data iting_grafana_data 2>/dev/null || true

# Clean old backups
find $BACKUP_DIR -name "*.tar.gz" -mtime +$RETENTION_DAYS -delete

echo "Backup completed: $TIMESTAMP"
ls -la $BACKUP_DIR/
BACKUPEOF

chmod +x /opt/iting/scripts/backup.sh

# Add daily backup cron
(crontab -l 2>/dev/null; echo "0 3 * * * /opt/iting/scripts/backup.sh >> /var/log/iting-backup.log 2>&1") | crontab -
```
